# Pinned to versions that ship prebuilt wheels for BOTH the local Python and
# the Python that Streamlit Community Cloud provisions (currently 3.14).
#
# Every version below is chosen because a cp314 manylinux wheel exists. If a
# package has no matching wheel, pip falls back to building from source — for
# the Rust/C extensions here that either takes ~10+ minutes or fails outright.
streamlit==1.58.0
# pydantic pins pydantic-core to an exact version itself, so do NOT pin
# pydantic-core separately — an independent pin can select a version with no
# wheel for the running interpreter, forcing a Rust build via maturin/pyo3.
# 2.12.5 resolves pydantic-core==2.41.5, the first line with cp314 wheels.
pydantic==2.12.5
pydantic-settings==2.14.2
anthropic==0.111.0
chromadb==1.5.9
# numpy is a transitive dep of chromadb. 2.3.2 is the earliest release with a
# cp314 wheel; 2.4.x shipped a broken/blocked C-extension DLL locally, so stay
# on the 2.3 line to keep both platforms healthy.
numpy==2.3.2
networkx==3.6.1
pygithub==2.9.1
gitpython==3.1.50

# Validation and Python security/dependency checks used by the pipeline.
pytest==8.3.5
bandit==1.8.3
pip-audit==2.9.0
